from selenium import webdriver
from selenium.webdriver.common.by import By

base_url = "https://batdongsan.com.vn/nha-dat-ban"
Links_ToDo = []

Links_ToDo.append(base_url)
for i in range(2, 2000):
    Link = f"{base_url}/p{i}"
    Links_ToDo.append(Link)

# Khởi tạo trình duyệt Chrome
driver = webdriver.Chrome()
driver.implicitly_wait(10)

news_urls = []

for url in Links_ToDo:
    driver.get(url)
    
    try:
        product_links = driver.find_elements(By.CSS_SELECTOR, "a.js__product-link-for-product-id")
        for link in product_links:
            new_url = link.get_attribute("href")
            news_urls.append(new_url)
    except Exception as e:
        print(f"Lỗi link")

    if news_urls:
        with open(r"C:\Users\NhậtNguyễn\Downloads\link.txt", "w", encoding="utf-8") as file:
            for new_url in news_urls:
                file.write(new_url + "\n")
        print(f"{len(news_urls)} links")
    else:
        print("Không có liên kết")

driver.quit()


with open("link.txt", "r", encoding="utf-8") as file:
    news_urls = file.readlines()

driver = webdriver.Chrome()
driver.implicitly_wait(10)

for index, news_url in enumerate(news_urls, start=1):
    news_url = news_url.strip()
    print("Đang scraping:", news_url)
    driver.get(news_url)
    time.sleep(5)
    page_source = driver.page_source

    soup = BeautifulSoup(page_source, "html.parser")

    title_element = soup.find("h1", class_="re__pr-title pr-title js__pr-title")
    if title_element:
        title = title_element.text.strip()
    else:
        print("Không có tiêu đề")
        continue

    address_element = soup.find("span", class_="re__pr-short-description js__pr-address")
    address = address_element.text.strip() if address_element else "Address not found"

    property_specs = {}
    spec_items = soup.select(".re__pr-specs-content-item")
    for item in spec_items:
        title_item = item.find("span", class_="re__pr-specs-content-item-title")
        value_item = item.find("span", class_="re__pr-specs-content-item-value")
        title_text = title_item.text.strip() if title_item else ""
        value_text = value_item.text.strip() if value_item else ""
        property_specs[title_text] = value_text

    filename = os.path.join(r"C:\Users\Downloads\data", f"article_{index}.txt")
    with open(filename, "w", encoding="utf-8") as f:
        f.write(f"Tiêu đề: {title}\n")
        f.write(f"Địa chỉ: {address}\n")
        f.write("Các thông tin bổ sung:\n")
        for key, value in property_specs.items():
            f.write(f"{key}: {value}\n")
        f.write("-" * 50 + "\n")
    print(f"Saved to {filename}")

driver.quit()
